The analysis in that file is genuinely correct. Every point lands. Let me add one thing before implementation:
The 60/40 is right but only if it's a pool model, not an individual rate. There's a big difference:
Wrong way (individual rate): Every creator gets 40% of downloads their models generate. This means if Creator A's models get downloaded 1000 times and Creator B's get downloaded 10 times, Creator A earns 40× more. Simple math, but it rewards whoever uploads the most popular models regardless of how much total revenue actually came in. Sarah could end up owing more than she collected.
Right way (pool model): 40% of total monthly subscription revenue goes into one pot. That pot gets split by download share. If total revenue is 300,000 DZD, 120,000 DZD goes to creators. Creator A had 70% of all downloads → gets 84,000 DZD. Creator B had 30% → gets 36,000 DZD. Sarah always keeps exactly 60%. Math never breaks.
Sarah's instinct is generous which is good for attracting creators. But the pool model protects her. Implement that.

phppublic function earnings(Request $request): JsonResponse
{
    $user        = $request->user();
    $poolPercent = (int) env('CREATOR_POOL_PERCENTAGE', 40);
    $currentMonth = now()->startOfMonth();

    // Total active subscriptions revenue this month
    $monthlyRevenue = \App\Models\Subscription::where('status', 'active')
        ->where('starts_at', '<=', now())
        ->where(function ($q) { $q->whereNull('ends_at')->orWhere('ends_at', '>', now()); })
        ->with('plan')
        ->get()
        ->sum(fn($sub) => $sub->plan->price_dzd / $sub->plan->duration_months);

    // Creator pool (40% of revenue)
    $creatorPool = round($monthlyRevenue * ($poolPercent / 100));

    // Platform keeps
    $platformKeeps = $monthlyRevenue - $creatorPool;

    // This creator's downloads this month
    $myModelIds = SketchupModel::where('creator_id', $user->id)->pluck('id');

    $myDownloads = Download::whereIn('sketchup_model_id', $myModelIds)
        ->where('delivered_at', '>=', $currentMonth)
        ->count();

    // Total platform downloads this month (all creator models only, not Sarah's)
    $allCreatorModelIds = SketchupModel::whereNotNull('creator_id')->pluck('id');

    $totalCreatorDownloads = Download::whereIn('sketchup_model_id', $allCreatorModelIds)
        ->where('delivered_at', '>=', $currentMonth)
        ->count();

    // This creator's share of the pool
    $mySharePercent  = $totalCreatorDownloads > 0
        ? round(($myDownloads / $totalCreatorDownloads) * 100, 2)
        : 0;

    $myEarningsDzd = $totalCreatorDownloads > 0
        ? round(($myDownloads / $totalCreatorDownloads) * $creatorPool)
        : 0;

    // Per model breakdown
    $perModel = Download::whereIn('sketchup_model_id', $myModelIds)
        ->where('delivered_at', '>=', $currentMonth)
        ->selectRaw('sketchup_model_id, count(*) as download_count')
        ->groupBy('sketchup_model_id')
        ->orderByDesc('download_count')
        ->get()
        ->map(function ($row) {
            $model = SketchupModel::find($row->sketchup_model_id);
            return [
                'model_id'       => $row->sketchup_model_id,
                'model_name'     => $model->name ?? 'Unknown',
                'download_count' => $row->download_count,
            ];
        });

    return response()->json([
        'paypal_email'             => $user->paypal_email,
        'platform_split'           => 100 - $poolPercent,   // 60
        'creator_pool_split'       => $poolPercent,          // 40
        'current_month'            => [
            'label'                    => now()->format('F Y'),
            'total_platform_revenue'   => $monthlyRevenue,
            'creator_pool_total'       => $creatorPool,
            'platform_keeps'           => $platformKeeps,
            'your_downloads'           => $myDownloads,
            'total_creator_downloads'  => $totalCreatorDownloads,
            'your_share_percent'       => $mySharePercent,
            'your_estimated_earnings'  => $myEarningsDzd,
        ],
        'top_models' => $perModel,
    ]);
}

<?php
namespace App\Filament\Pages;

use App\Models\Download;
use App\Models\Subscription;
use App\Models\SketchupModel;
use App\Models\User;
use Filament\Pages\Page;

class CreatorEarningsOverview extends Page
{
    protected static ?string $navigationIcon  = 'heroicon-o-banknotes';
    protected static ?string $navigationGroup = 'Creators';
    protected static ?string $navigationLabel = 'Monthly payouts';
    protected static string  $view            = 'filament.pages.creator-earnings-overview';

    public array $creatorBreakdown = [];
    public int   $totalRevenue     = 0;
    public int   $creatorPool      = 0;
    public int   $platformKeeps    = 0;
    public string $month           = '';

    public function mount(): void
    {
        $poolPercent  = (int) env('CREATOR_POOL_PERCENTAGE', 40);
        $currentMonth = now()->startOfMonth();
        $this->month  = now()->format('F Y');

        $this->totalRevenue = (int) Subscription::where('status', 'active')
            ->where('starts_at', '<=', now())
            ->where(fn($q) => $q->whereNull('ends_at')->orWhere('ends_at', '>', now()))
            ->with('plan')
            ->get()
            ->sum(fn($s) => $s->plan->price_dzd / $s->plan->duration_months);

        $this->creatorPool  = (int) round($this->totalRevenue * ($poolPercent / 100));
        $this->platformKeeps = $this->totalRevenue - $this->creatorPool;

        $allCreatorModelIds = SketchupModel::whereNotNull('creator_id')->pluck('id');

        $totalDownloads = Download::whereIn('sketchup_model_id', $allCreatorModelIds)
            ->where('delivered_at', '>=', $currentMonth)
            ->count();

        $creators = User::where('is_creator', true)
            ->where('creator_status', 'approved')
            ->get();

        $this->creatorBreakdown = $creators->map(function ($creator) use (
            $currentMonth, $totalDownloads, $allCreatorModelIds
        ) {
            $myIds = SketchupModel::where('creator_id', $creator->id)->pluck('id');

            $myDownloads = Download::whereIn('sketchup_model_id', $myIds)
                ->where('delivered_at', '>=', $currentMonth)
                ->count();

            $sharePercent = $totalDownloads > 0
                ? round(($myDownloads / $totalDownloads) * 100, 1)
                : 0;

            $earningsDzd = $totalDownloads > 0
                ? (int) round(($myDownloads / $totalDownloads) * $this->creatorPool)
                : 0;

            return [
                'name'         => $creator->display_name ?? $creator->name,
                'email'        => $creator->email,
                'paypal'       => $creator->paypal_email ?? '—',
                'downloads'    => $myDownloads,
                'share'        => $sharePercent,
                'payout_dzd'   => $earningsDzd,
            ];
        })
        ->sortByDesc('downloads')
        ->values()
        ->toArray();
    }
}
